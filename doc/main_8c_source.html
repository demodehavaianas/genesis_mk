78int main(bool hardReset)
79{
80 engine_boot();
81
82 debugEnabled = FALSE;
83 gRoom = PALACE_GATES;
84 gFrames = 0;
85 gInd_tileset = TILE_USER_INDEX;
86 player[0].id = JOHNNY_CAGE;
87
88 if (!hardReset)
89 {
90 SYS_hardReset();
91 }
92
93 resetGraphicElements();
94
95 while (TRUE)
96 {
97 inputSystem();
98
99 if (TELA_DEMO_INTRO == gRoom)
100 {
101 processIntro();
102 }
103
104 if (TELA_START == gRoom)
105 {
106 processPressStart();
107 }
108
109 if (SELECAO_PERSONAGENS == gRoom)
110 {
111 processSelecaoPersonagens();
112 }
113
114 if (BONUS_STAGE == gRoom)
115 {
116 processBonusStage();
117 }
118
119 if (PALACE_GATES == gRoom)
120 {
121 initPalaceGatesRoom();
122 }
123
124 if (debugEnabled)
125 {
126 char str[64];
127 sprintf(str, "tiles nos BGs: %d", gInd_tileset);
128 VDP_drawText(str, 1, 1);
129 sprintf(str, "contagem de frames: %ld", gFrames);
130 VDP_drawText(str, 1, 2);
131 }
132
133 // -- FINALIZAÇÕES -- //
134 // VDP_showFPS(1, 1, 1);
135 SPR_update(); // Atualização dos sprites na tela
136 SYS_doVBlankProcess(); // Sincroniza com o VBlank
137 }
138
139 return 0;
140}
